name: Build NetSpoof APK

on:
  push:
    branches: [ main, master ]
  workflow_dispatch:

jobs:
  build:
    runs-on: ubuntu-latest
    steps:

    - name: Checkout code
      uses: actions/checkout@v4

    - name: Find and extract ZIP
      run: |
        ZIP_FILE=$(find . -name "*.zip" -not -path "*/\.*" | head -1)
        if [ -z "$ZIP_FILE" ]; then
          echo "ERROR: Repo mein koi .zip file nahi mili!"
          echo "Uploaded files:"
          ls -la
          exit 1
        fi
        echo "Found ZIP: $ZIP_FILE"
        unzip -o "$ZIP_FILE" -d extracted_project
        echo "Contents:"
        ls extracted_project/
        GRADLE_FILE=$(find extracted_project -name "gradlew" | head -1)
        if [ -z "$GRADLE_FILE" ]; then
          echo "ERROR: gradlew not found in ZIP!"
          find extracted_project -type f | head -30
          exit 1
        fi
        PROJECT_DIR=$(dirname "$GRADLE_FILE")
        echo "Project root: $PROJECT_DIR"
        echo "PROJECT_DIR=$PROJECT_DIR" >> $GITHUB_ENV

    - name: Set up JDK 17
      uses: actions/setup-java@v4
      with:
        java-version: '17'
        distribution: 'temurin'

    - name: Set up Android SDK
      uses: android-actions/setup-android@v3

    - name: Build APK
      run: |
        cd "$PROJECT_DIR"
        chmod +x gradlew
        ./gradlew assembleDebug --no-daemon --stacktrace

    - name: Upload APK
      uses: actions/upload-artifact@v4
      with:
        name: NetSpoof-APK
        path: "**/outputs/apk/debug/*.apk"
        retention-days: 30
